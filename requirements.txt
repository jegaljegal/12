# 3. Gemini AI 도우미 함수
# ==========================================
def ask_gemini_ai(prompt_text):
    """Gemini API를 사용하여 교실 위치 및 이동 관련 질문에 답변"""
    try:
        from google import genai
        
        # Streamlit Secrets 또는 환경 변수에서 API 키 확인
        api_key = st.secrets.get("GEMINI_API_KEY") or os.environ.get("GEMINI_API_KEY")
        if not api_key:
            return "⚠️ Gemini API 키가 설정되지 않았습니다. Secrets에 GEMINI_API_KEY를 등록해주세요."
        
        client = genai.Client(api_key=api_key)
        
        system_context = f"""
        너는 학교 교실 이동 및 시간표 안내 AI 도우미야.
        학생들이 수업 장소를 헷갈리지 않게 친절하고 명확하게 안내해줘.
        다음 교실 위치 정보를 참고해서 답변해:
        {ROOM_DB}
        
        오늘 1학년 1반 변경사항: 3교시 과학실1, 4교시 체육관, 6교시 컴퓨터실
        오늘 1학년 2반 변경사항: 2교시 음악실, 5교시 미술실
        """
        
        response = client.models.generate_content(
            model="gemini-2.5-flash-lite",
            contents=f"{system_context}\n\n사용자 질문: {prompt_text}"
        )
        return response.text
    except Exception as e:
        return f"❌ AI 답변을 불러오는 중 오류가 발생했습니다: {str(e)}"
